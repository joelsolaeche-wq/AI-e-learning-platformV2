  <read_first>
    - .planning/phases/01-auth-rls-foundation/01-RESEARCH.md — "components/theme-provider.tsx" and "app/layout.tsx — forced dark" sections with exact code
  </read_first>
  <action>
    **Step 1 — Create `components/theme-provider.tsx`:**

    ```tsx
    'use client'

    import { ThemeProvider as NextThemesProvider } from 'next-themes'

    export function ThemeProvider({ children, ...props }: React.ComponentProps<typeof NextThemesProvider>) {
      return <NextThemesProvider {...props}>{children}</NextThemesProvider>
    }
    ```

    **Step 2 — Overwrite `app/layout.tsx`:**

    ```tsx
    import type { Metadata } from 'next'
    import { Inter } from 'next/font/google'
    import { ThemeProvider } from '@/components/theme-provider'
    import './globals.css'

    const inter = Inter({ subsets: ['latin'] })

    export const metadata: Metadata = {
      title: 'AI Learning Platform',
      description: 'Cohort-based AI learning for enterprise teams',
    }

    export default function RootLayout({
      children,
    }: {
      children: React.ReactNode
    }) {
      return (
        <html lang="en" suppressHydrationWarning>
          <body className={inter.className}>
            <ThemeProvider
              attribute="class"
              defaultTheme="dark"
              forcedTheme="dark"
              disableTransitionOnChange
            >
              {children}
            </ThemeProvider>
          </body>
        </html>
      )
    }
    ```

    Key constraints enforced:
    - `app/layout.tsx` has NO `"use client"` directive — it remains a Server Component
    - `ThemeProvider` is the only Client Component wrapper — it is imported from `@/components/theme-provider`
    - `forcedTheme="dark"` ensures the OS preference is ignored entirely
    - `suppressHydrationWarning` on `<html>` prevents the React mismatch warning from `next-themes` adding the `dark` class on the client
  </action>
  <acceptance_criteria>

      'NEXT_PUBLIC_SUPABASE_URL',
