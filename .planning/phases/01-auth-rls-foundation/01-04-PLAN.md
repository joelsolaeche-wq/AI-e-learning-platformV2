    ```ts
    import { createServerClient } from '@supabase/ssr'
    import { NextResponse, type NextRequest } from 'next/server'

    /**
     * Next.js Middleware — runs on every matched request BEFORE the route renders.
     *
     * Two responsibilities:
     * 1. Session refresh: creates a Supabase client that reads/writes the session
     *    cookie on every request, keeping sessions alive (AUTH-02).
     * 2. Route protection: redirects unauthenticated users away from protected
     *    paths and authenticated users away from auth pages (AUTH-04, AUTH-02).
     *
     * CRITICAL RULES (see research pitfalls):
     * - Use getUser() not getSession() — getUser() validates the JWT server-side.
     *   getSession() reads the cookie without verification (security gap).
     * - Always return supabaseResponse, not a freshly constructed NextResponse.
     *   The Supabase client writes the refreshed cookie onto supabaseResponse.
     *   Returning a different response object loses the refreshed cookie and
     *   breaks AUTH-02 (sessions expire mid-visit).
     */
    export async function middleware(request: NextRequest) {
      // supabaseResponse is the canonical response object for this request.
      // All cookie writes from the Supabase client land on this object.
      // Never discard it — always return it or forward its cookies.
      let supabaseResponse = NextResponse.next({ request })

      const supabase = createServerClient(
        process.env.NEXT_PUBLIC_SUPABASE_URL!,
        process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
        {
          cookies: {
            getAll() {
              return request.cookies.getAll()
            },
            setAll(cookiesToSet) {
              // Step 1: write cookies onto the request so subsequent middleware
              // or route handlers see the updated values within this request.
              cookiesToSet.forEach(({ name, value }) =>
                request.cookies.set(name, value)
              )
              // Step 2: reconstruct supabaseResponse from the updated request
              // so the response carries the updated cookies to the browser.
              supabaseResponse = NextResponse.next({ request })
              cookiesToSet.forEach(({ name, value, options }) =>
                supabaseResponse.cookies.set(name, value, options)
              )
            },
          },
        }
      )

      // IMPORTANT: use getUser(), NOT getSession().
      // getSession() does not contact the Supabase Auth server and cannot
      // detect revoked sessions. getUser() validates the JWT server-side.
      const {
        data: { user },
      } = await supabase.auth.getUser()

      const { pathname } = request.nextUrl

      // ----------------------------------------------------------------
      // Protected route guard (AUTH-04)
      // Paths that require authentication. Extend this list as new
      // protected sections are added in later phases.
      // ----------------------------------------------------------------
      const protectedPaths = ['/dashboard', '/cohorts', '/lessons', '/admin']
      const isProtected = protectedPaths.some((p) => pathname.startsWith(p))

      if (!user && isProtected) {
        const loginUrl = request.nextUrl.clone()
        loginUrl.pathname = '/auth/login'
        // Preserve the intended destination for post-login redirect
        loginUrl.searchParams.set('redirectTo', pathname)
        return NextResponse.redirect(loginUrl)
      }

      // ----------------------------------------------------------------
      // Auth page redirect for authenticated users (AUTH-02)
      // Authenticated users visiting /auth/login or /auth/register are
      // sent to /dashboard — they are already logged in.
      // ----------------------------------------------------------------
      const authPaths = ['/auth/login', '/auth/register']
      const isAuthPage = authPaths.some((p) => pathname.startsWith(p))

      if (user && isAuthPage) {
        const dashboardUrl = request.nextUrl.clone()
        dashboardUrl.pathname = '/dashboard'
        return NextResponse.redirect(dashboardUrl)
      }

      // ----------------------------------------------------------------
      // Pass through — return supabaseResponse with refreshed session cookie.
      // This is the ONLY valid way to end middleware when not redirecting.
      // Returning NextResponse.next() here would lose the refreshed cookie.
      // ----------------------------------------------------------------
      return supabaseResponse
    }

    /**
     * Matcher config: run middleware on every request EXCEPT:
     * - Next.js static files (_next/static, _next/image)
     * - favicon.ico
     * - Static asset extensions (svg, png, jpg, etc.)
     *
     * This ensures session refresh happens on every page navigation,
     * which is required for AUTH-02 (stay logged in across sessions).
     */
    export const config = {
      matcher: [
        '/((?!_next/static|_next/image|favicon.ico|.*\\.(?:svg|png|jpg|jpeg|gif|webp)$).*)',
      ],
    }
    ```

    Create `app/dashboard/page.tsx`:

    ```tsx
    import { createClient } from '@/lib/supabase/server'
    import { redirect } from 'next/navigation'

    /**
     * Dashboard Page — protected Server Component.
     *
     * The middleware already guards this route (unauthenticated users are
     * redirected to /auth/login before this component runs). The getUser()
     * call here is a belt-and-suspenders check — it ensures that even if
     * middleware is misconfigured, unauthenticated users cannot see this page.
     *
     * Uses the server client (not the browser client) for the getUser() call
     * because this is a Server Component.
     */
    export default async function DashboardPage() {
      const supabase = await createClient()
      const {
        data: { user },
      } = await supabase.auth.getUser()

      // Belt-and-suspenders: middleware should have redirected already,
      // but this guard ensures the page is never rendered without a session.
      if (!user) {
        redirect('/auth/login')
      }

      return (
        <main className="flex min-h-screen flex-col p-8">
          <div className="mx-auto w-full max-w-4xl space-y-8">
            <header className="flex items-center justify-between">
              <div>
                <h1 className="text-3xl font-bold tracking-tight">Dashboard</h1>
                <p className="mt-1 text-muted-foreground">
                  Welcome back, {user.email}
                </p>
              </div>

              {/* Logout button — POST to /auth/logout route handler */}
              <form action="/auth/logout" method="POST">
                <button
                  type="submit"
                  className="rounded-md bg-secondary px-4 py-2 text-sm font-medium text-secondary-foreground transition-colors hover:bg-secondary/80"
                >
                  Log out
                </button>
              </form>
            </header>

            <section className="rounded-lg border border-border bg-card p-6">
              <h2 className="text-lg font-semibold">Your learning journey</h2>
              <p className="mt-2 text-sm text-muted-foreground">
                Your enrolled cohorts will appear here. Cohort enrollment is
                coming in Phase 2.
              </p>
            </section>

            <section className="rounded-lg border border-border bg-card p-6">
              <h2 className="text-lg font-semibold">Account</h2>
              <dl className="mt-3 space-y-2 text-sm">
                <div className="flex gap-2">
                  <dt className="text-muted-foreground">Email:</dt>
                  <dd>{user.email}</dd>
                </div>
                <div className="flex gap-2">
                  <dt className="text-muted-foreground">User ID:</dt>
                  <dd className="font-mono text-xs">{user.id}</dd>
                </div>
              </dl>
            </section>
          </div>
        </main>
      )
    }
    ```

    Key constraints:
    - This is a Server Component (no `"use client"`)
    - Uses `createClient` from `@/lib/supabase/server` (not the browser client)
    - The logout button is a native `<form>` with `method="POST"` targeting `/auth/logout`
    - No shadcn/ui Button import needed for the logout — a native button keeps this page dependency-light and avoids making the layout a Client Component
  </action>

  <title>Verify session persistence across browser restart (AUTH-02)</title>
  <read_first>
    - .planning/ROADMAP.md — Phase 1 Success Criteria 2: "A returning user who closes and reopens the browser is still logged in"
    - .planning/phases/01-auth-rls-foundation/01-RESEARCH.md — "Pitfall 3: Not returning the supabaseResponse object" — this is what makes sessions persist
  </read_first>
  <action>
    This test verifies that the middleware correctly refreshes and persists the session cookie so that AUTH-02 (stay logged in across sessions) is satisfied.

    **Test steps:**

    1. Sign in to the app via `http://localhost:3000/auth/login`
    2. Verify you reach `/dashboard`
    3. Open DevTools → Application → Cookies → `http://localhost:3000`
    4. Note the `sb-*-auth-token` cookie — check that it has:
       - `HttpOnly`: true
       - `SameSite`: Lax
       - An expiry date in the future (not a session cookie — it should persist)
    5. Close the browser tab completely
    6. Reopen the browser and navigate to `http://localhost:3000/dashboard`
    7. Expected: Dashboard loads directly — no redirect to login

    **If the session does NOT persist (redirected to login after reopen):**

    Check these three things in `middleware.ts`:
    1. Is `return supabaseResponse` at the end (not `return NextResponse.next()`)?
    2. Does `setAll` call `supabaseResponse = NextResponse.next({ request })` to rebuild with updated cookies?
    3. Is `supabase.auth.getUser()` being awaited before the return?

    Any one of these being wrong will cause the session cookie refresh to be lost.

    **Automated check — cookie expiry:**

    ```bash
    # Check that the Supabase cookie is not a session-only cookie (has Max-Age or Expires)
    curl -s -c /tmp/cookies.txt http://localhost:3000/dashboard
    grep "sb-" /tmp/cookies.txt
    # Expected: lines with non-zero expiry timestamps
    ```
  </action>
  <acceptance_criteria>
    - Supabase session cookie has a future expiry (not session-only)
    - After closing and reopening the browser, navigating to `/dashboard` loads the page without redirect
    - `middleware.ts` ends with `return supabaseResponse` (not `NextResponse.next()`)
    - No TypeScript errors: `npx tsc --noEmit` exits 0
  </acceptance_criteria>
