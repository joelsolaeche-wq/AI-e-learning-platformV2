```typescript
'use server'

import { createClient } from '@/lib/supabase/server'
import { redirect } from 'next/navigation'
import { revalidatePath } from 'next/cache'

export type EnrollmentActionResult = {
  error: string | null
}

/**
 * Enrolls the current user in a cohort.
 *
 * Per CONTEXT.md D-01: writes an enrollments row, then redirects to /dashboard.
 * Per CONTEXT.md D-03: a duplicate-enrollment error (Postgres 23505) is treated
 * as success (the user is already enrolled) and we redirect silently.
 *
 * Form contract:
 *   <form action={enrollInCohortAction}>
 *     <input type="hidden" name="cohort_id" value={cohort.id} />
 *     <button type="submit">Join Cohort</button>
 *   </form>
 *
 * The action is callable as a plain form action OR via useFormState — both
 * shapes are accommodated by the (_prevState, formData) signature.
 */
export async function enrollInCohortAction(
  _prevState: EnrollmentActionResult,
  formData: FormData
): Promise<EnrollmentActionResult> {
  const cohortId = formData.get('cohort_id') as string | null

  if (!cohortId) {
    return { error: 'Cohort ID is required.' }
  }

  const supabase = await createClient()

  // Auth guard — never trust the caller
  const {
    data: { user },
  } = await supabase.auth.getUser()
  if (!user) {
    redirect('/auth/login')
  }

  const { error } = await supabase
    .from('enrollments')
    .insert({ user_id: user.id, cohort_id: cohortId })

  if (error) {
    // Postgres 23505 = unique_violation. Per D-03, an existing enrollment is not
    // an error from the user's perspective — silently send them to the dashboard.
    if (error.code === '23505') {
      revalidatePath('/dashboard')
      redirect('/dashboard')
    }
    // Any other failure: bubble a friendly toast string per UI-SPEC copywriting.
    return { error: "Couldn't enroll — try again" }
  }

  revalidatePath('/dashboard')
  redirect('/dashboard')
}
```

Important notes:
- Use the SAME signature shape as `signInAction` from auth.actions.ts: `(_prevState, formData) => Promise<EnrollmentActionResult>`. This allows the action to be wired with `useFormState` later if needed; for Phase 3 it's used as a bare form action which Next.js calls with `(undefined, formData)` — the `_prevState` is ignored.
- The exact error string `"Couldn't enroll — try again"` is locked by UI-SPEC; do NOT paraphrase.
- DO NOT add any logging, telemetry, or extra branches. Match the auth.actions.ts level of minimalism.
- Per D-02: no client component is needed. This file has `'use server'` at the top — same convention as auth.actions.ts.

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser → Server Action | Form data (cohort_id) crosses from untrusted client to server |
| Server Action → Postgres | RLS-bounded write; the `with check (auth.uid() = user_id)` policy is the final gate |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-05 | S (Spoofing) | enrollInCohortAction caller identity | mitigate | The action calls `supabase.auth.getUser()` and uses `user.id` server-side — the client cannot forge the user_id; only the cohort_id is read from the form |
| T-03-06 | T (Tampering) | Client-supplied cohort_id | mitigate | The Server Action does NOT validate that cohort_id is published or active — it relies on the FK constraint (`cohort_id REFERENCES public.cohorts(id)`) to reject non-existent IDs and on the unique constraint to dedupe; this is acceptable because cohorts are read-only via RLS and an attacker enrolling in a junk cohort harms only their own dashboard |
| T-03-07 | E (Elevation) | Server Action could be called by unauthenticated user | mitigate | Auth guard at top of action: `if (!user) redirect('/auth/login')`. RLS on enrollments INSERT is a second line of defence (`with check (auth.uid() = user_id)`) |
| T-03-08 | I (Information disclosure) | enrollmentsResult query in page | accept | The page only selects `cohort_id` for the current user — no other user's enrollments are read; existing SELECT RLS already enforces this |
| T-03-09 | D (Denial of service) | Spamming the Server Action with duplicate inserts | accept | UNIQUE constraint dedupes at DB layer; 23505 path returns instantly with no extra work; rate limiting deferred to platform layer |
</threat_model>
