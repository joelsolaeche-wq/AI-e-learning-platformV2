```typescript
'use client'

import { useCallback, useRef } from 'react'
import MuxPlayer from '@mux/mux-player-react'

interface VideoPlayerProps {
  playbackId: string
  lessonId: string
  resumePosition: number
  duration: number
}

export function VideoPlayer({ playbackId, lessonId, resumePosition, duration }: VideoPlayerProps) {
  const lastSaveRef = useRef<number>(0)
  const completedRef = useRef<boolean>(false)

  const saveProgress = useCallback(async (position: number, completed: boolean) => {
    try {
      await fetch('/api/video/progress', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ lessonId, position, duration, completed }),
      })
    } catch {
      // progress save failures are non-fatal
    }
  }, [lessonId, duration])

  const handleTimeUpdate = useCallback((event: Event) => {
    const target = event.target as HTMLVideoElement
    const pos = Math.floor(target.currentTime)
    const now = Date.now()
    const pct = duration > 0 ? pos / duration : 0

    if (pct >= 0.9 && !completedRef.current) {
      completedRef.current = true
      saveProgress(pos, true)   // <-- fires once; no re-render triggered
      return
    }

    if (now - lastSaveRef.current >= 10_000) {
      lastSaveRef.current = now
      saveProgress(pos, false)
    }
  }, [duration, saveProgress])

  const handleEnded = useCallback(() => {
    saveProgress(duration, true)
  }, [duration, saveProgress])

  return (
    <MuxPlayer
      playbackId={playbackId}
      startTime={resumePosition}
      onTimeUpdate={handleTimeUpdate as unknown as () => void}
      onEnded={handleEnded}
      style={{ width: '100%', aspectRatio: '16/9' }}
      accentColor="#ffffff"
      defaultShowRemainingTime
      playbackRates={[0.75, 1, 1.25, 1.5, 2]}
    />
  )
}
```

  <action>
The fix is surgical — two lines change, one import added.

**What to do:**

1. Add `useRouter` to the import from `next/navigation` (new import line after the existing React import):
   ```typescript
   import { useRouter } from 'next/navigation'
   ```

2. Inside the `VideoPlayer` function body, after the `completedRef` declaration, call the hook:
   ```typescript
   const router = useRouter()
   ```

3. `saveProgress` currently returns `void` (the fetch result is discarded). To call `router.refresh()` only after the API confirms completion, convert `saveProgress` to return `Promise<void>` and await it at the call site in `handleTimeUpdate`.

   Change the `handleTimeUpdate` completion branch from:
   ```typescript
   if (pct >= 0.9 && !completedRef.current) {
     completedRef.current = true
     saveProgress(pos, true)
     return
   }
   ```
   to:
   ```typescript
   if (pct >= 0.9 && !completedRef.current) {
     completedRef.current = true
     saveProgress(pos, true).then(() => router.refresh())
     return
   }
   ```

   `handleTimeUpdate` is synchronous (MuxPlayer fires it many times per second), so use `.then()` rather than making the callback async — making it async would not change behavior but adds unnecessary complexity. The `.then()` chain means `router.refresh()` fires only after the fetch resolves successfully, preventing a re-render race with an in-flight DB write.

4. `handleEnded` also calls `saveProgress(duration, true)`. Apply the same pattern there:
   ```typescript
   const handleEnded = useCallback(() => {
     saveProgress(duration, true).then(() => router.refresh())
   }, [duration, saveProgress, router])
   ```
   Add `router` to the `useCallback` dependency array.

5. The `useCallback` for `handleTimeUpdate` must also list `router` in its deps array:
   ```typescript
   }, [duration, saveProgress, router])
   ```

**What NOT to do:**
- Do not add `router` to `saveProgress`'s own `useCallback` — saveProgress is a shared helper; the router call belongs at the call site.
- Do not set `completedRef.current = true` inside `.then()` — it must remain before the async call so repeated `onTimeUpdate` firings do not trigger multiple fetches while the first is in-flight.
- Do not use `router.push()` or `router.replace()` — only `router.refresh()`. A push would navigate away; refresh re-renders Server Components in place.
