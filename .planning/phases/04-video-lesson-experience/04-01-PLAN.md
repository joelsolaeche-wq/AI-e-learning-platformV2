<action>
Create `components/VideoPlayer.tsx` as a Client Component.

The component receives:
- `playbackId: string` — Mux playback ID
- `lessonId: string` — for progress API calls
- `resumePosition: number` — seconds (from lesson_progress.last_position)
- `duration: number` — total video duration in seconds

Key behaviors:
1. Render `<MuxPlayer>` from `@mux/mux-player-react` with `startTime={resumePosition}`
2. Throttle `onTimeUpdate` to fire the progress API at most every 10 seconds
3. When position/duration >= 0.90 for the first time, mark `completed=true` locally and fire the API immediately with completed signal
4. `onEnded` also fires the API with final position

Progress API call (via fetch POST to `/api/video/progress`):
```
body: JSON.stringify({ lessonId, position: Math.floor(currentTime), duration })
```

Throttle implementation — use a ref to track last save time:
```tsx
const lastSaveRef = useRef<number>(0)
const completedRef = useRef<boolean>(false)

const saveProgress = useCallback(async (position: number, isCompleted: boolean) => {
  await fetch('/api/video/progress', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ lessonId, position, duration, completed: isCompleted }),
  })
}, [lessonId, duration])

const handleTimeUpdate = useCallback((event: Event) => {
  const target = event.target as HTMLVideoElement
  const pos = Math.floor(target.currentTime)
  const now = Date.now()
  const pct = duration > 0 ? pos / duration : 0

  if (pct >= 0.9 && !completedRef.current) {
    completedRef.current = true
    saveProgress(pos, true)
    return
  }

  if (now - lastSaveRef.current >= 10_000) {
    lastSaveRef.current = now
    saveProgress(pos, false)
  }
}, [duration, saveProgress])
```

MuxPlayer props:
- `playbackId={playbackId}`
- `startTime={resumePosition}`
- `onTimeUpdate={handleTimeUpdate}`
- `onEnded={() => saveProgress(Math.floor(duration), completedRef.current)}`
- `style={{ width: '100%', aspectRatio: '16/9' }}`
- `accentColor="#ffffff"`
- `defaultShowRemainingTime`
- `playbackRates={[0.75, 1, 1.25, 1.5, 2]}` — enables speed selector in the player UI

Full file:
```tsx
'use client'

import { useCallback, useRef } from 'react'
import MuxPlayer from '@mux/mux-player-react'

interface VideoPlayerProps {
  playbackId: string
  lessonId: string
  resumePosition: number
  duration: number
}

export function VideoPlayer({ playbackId, lessonId, resumePosition, duration }: VideoPlayerProps) {
  const lastSaveRef = useRef<number>(0)
  const completedRef = useRef<boolean>(false)

  const saveProgress = useCallback(async (position: number, completed: boolean) => {
    try {
      await fetch('/api/video/progress', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ lessonId, position, duration, completed }),
      })
    } catch {
      // progress save failures are non-fatal
    }
  }, [lessonId, duration])

  const handleTimeUpdate = useCallback((event: Event) => {
    const target = event.target as HTMLVideoElement
    const pos = Math.floor(target.currentTime)
    const now = Date.now()
    const pct = duration > 0 ? pos / duration : 0

    if (pct >= 0.9 && !completedRef.current) {
      completedRef.current = true
      saveProgress(pos, true)
      return
    }

    if (now - lastSaveRef.current >= 10_000) {
      lastSaveRef.current = now
      saveProgress(pos, false)
    }
  }, [duration, saveProgress])

  const handleEnded = useCallback(() => {
    saveProgress(duration, true)
  }, [duration, saveProgress])

  return (
    <MuxPlayer
      playbackId={playbackId}
      startTime={resumePosition}
      onTimeUpdate={handleTimeUpdate as EventListenerOrEventListenerObject}
      onEnded={handleEnded}
      style={{ width: '100%', aspectRatio: '16/9' }}
      accentColor="#ffffff"
      defaultShowRemainingTime
      playbackRates={[0.75, 1, 1.25, 1.5, 2]}
    />
  )
}
```

This Server Component:
1. Auth-checks the user (redirect to /auth/login if not authenticated)
2. Fetches the lesson row (id, title, module_id, mux_playback_id, duration_seconds, transcript)
3. Fetches lesson_progress for (user.id, lessonId) — gets last_position for resume
4. RLS on lessons already restricts access to enrolled users — if lesson returns null (not found or not enrolled), call notFound()
5. Renders a centered layout with: lesson title, VideoPlayer component, transcript section

Route params: `{ lessonId: string }` — accessed via `await params`

Type aliases (follow existing Pick pattern from other pages):
```tsx
type LessonRow = Pick<
  Database['public']['Tables']['lessons']['Row'],
  'id' | 'title' | 'module_id' | 'mux_playback_id' | 'duration_seconds' | 'transcript'
>
type ProgressRow = Pick<
  Database['public']['Tables']['lesson_progress']['Row'],
  'last_position' | 'completed'
>
```

Data fetching (parallel):
```tsx
const [lessonResult, progressResult] = await Promise.all([
  supabase
    .from('lessons')
    .select('id, title, module_id, mux_playback_id, duration_seconds, transcript')
    .eq('id', lessonId)
    .single(),
  supabase
    .from('lesson_progress')
    .select('last_position, completed')
    .eq('user_id', user.id)
    .eq('lesson_id', lessonId)
    .maybeSingle(),
])
```

If lessonResult.error or !lessonResult.data → notFound()
If !lesson.mux_playback_id → render a "Video not available" placeholder instead of VideoPlayer

resumePosition = progressResult.data?.last_position ?? 0

Layout:
```tsx
return (
```
